#
# GSP <-> driver shared message queue injection.
#
# Shared buffer layout, from message_queue_cpu.c:_getMsgQueueParams() and
# GspMsgQueuesInit().  On silicon commandQueueSize = statusQueueSize = 0x40000
# and pageTableSize rounds up to 0x1000, giving sharedBufSize = 0x81000:
#
#   [0x00000, 0x01000)  page table   - PTEs the GSP uses to map the buffer
#   [0x01000, 0x41000)  command queue - driver -> GSP   (GSP reads this)
#   [0x41000, 0x81000)  status queue  - GSP -> driver   (driver reads this)
#
# Only the status queue is in scope: it is the sink the *host driver* parses.
# The page table and the command queue are GSP-side inputs; corrupting them
# faults the GSP (Xid 120) and takes the device down, which stops the fuzzer
# rather than exercising the driver.  Every call below is therefore anchored
# at the status queue base and cannot reach the other two regions.
#
# Status queue internals, from msgq.c:msgqTxCreate() called with
# msgSize = GSP_MSG_QUEUE_ELEMENT_SIZE_MIN = 0x1000, hdrAlign = 4,
# entryAlign = 12, size = 0x40000:
#
#   +0x00000  msgqTxHeader  version, size, msgSize, msgCount, writePtr,
#                           flags, rxHdrOff, entryOff   (writePtr at +0x10)
#   +0x00020  msgqRxHeader  readPtr
#   +0x01000  msgCount = (0x40000 - 0x1000) / 0x1000 = 63 entries of 0x1000
#
# msgqRxLink() snapshots the whole msgqTxHeader into driver-private memory
# once, at attach time, and validates it there (msgq.c:315).  After that the
# only word the driver re-reads live from the status queue header is
# writePtr, via pWriteIncoming in msgqRxGetReadAvailable() (msgq.c:600).
#
# All three calls are `mixed`:
#   dynamic - the status queue base address is read live per call from
#             /proc/driver/nvidia/gpus/*/instrumentation (utilities.cpp);
#             syz_mixed_gpu_sq_element additionally computes checkSum live.
#   static  - every argument range below is a compile-time fact about the
#             layout above.
#

# Raw write at `offset` bytes past the status queue base.
# static:  offset is confined to [0, statusQueueSize); the write is clamped in
#          the harness so offset+size never leaves the status queue.
# dynamic: absolute address = status_queue_offset + offset.
syz_mixed_gpu_sq_write(offset intptr[0:0x3ffff], buffer ptr[in, array[int8]], size bytesize[buffer])

# Publish a value into msgqTxHeader.writePtr at status queue +0x10.
# static:  [0:62].  msgqRxGetReadAvailable() returns 0 immediately when
#          writePtr >= msgCount (63), so values above 62 are a provable no-op
#          on the read path; they remain reachable via syz_mixed_gpu_sq_write
#          at offset 0x10 if a run ever needs them.
# dynamic: absolute address = status_queue_offset + 0x10.
syz_mixed_gpu_sq_set_write_ptr(write_ptr int32[0:62])

# Write one whole GSP_MSG_QUEUE_ELEMENT into entry slot `slot`.
# static:  slot [0:62] (msgCount); the element is exactly
#          GSP_MSG_QUEUE_ELEMENT_SIZE_MIN = 0x1000 bytes, matching the stride
#          msgqRxGetReadBuffer() hands to GspMsgQueueReceiveStatus().
# dynamic: absolute address = status_queue_offset + 0x1000 + slot * 0x1000.
#          When fix_checksum is 1 the harness recomputes the checkSum field
#          the way message_queue_priv.h:_checkSum32() does, so the element
#          survives the "Bad checksum" reject in GspMsgQueueReceiveStatus();
#          this needs rpc.length in range, so fix_checksum also folds length
#          into [32, 4048].  With fix_checksum 0 every byte is written verbatim.
syz_mixed_gpu_sq_element(slot int32[0:62], element ptr[in, gsp_msg_queue_element], fix_checksum bool32)

# GSP_MSG_QUEUE_ELEMENT from message_queue_priv.h:43, followed by
# rpc_message_header_v from g_rpc-message-header.h:41.  Field boundaries are
# declared so the mutator flips whole fields; no field value is restricted.
gsp_msg_queue_element {
	authTagBuffer	array[int8, 16]
	aadBuffer	array[int8, 16]
	checkSum	int32
	seqNum		int32
	elemCount	int32
# 4 bytes of tail padding: rpc is NV_DECLARE_ALIGNED(..., 8) at offset 48.
# The driver checksums these bytes, so leave them fuzzer-controlled.
	align_pad	int32
	rpc		rpc_message_header
	rpc_payload	array[int8, 4016]
} [packed]

rpc_message_header {
	header_version		int32
	signature		int32
	length			int32
	function		int32
	rpc_result		int32
	rpc_result_private	int32
	sequence		int32
	u			int32
} [packed]
